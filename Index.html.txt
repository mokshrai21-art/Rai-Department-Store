  <script>
    const products = [
      { id: 1, name: "Fortune Sunflower Oil (5L)", price: 750, img: "https://images.unsplash.com/photo-1474979266404-7eaacbcd87c5?w=200" },
      { id: 2, name: "Aashirvaad Atta (10kg)", price: 440, img: "https://images.unsplash.com/photo-1509440159596-0249088772ff?w=200" },
      { id: 3, name: "Basmati Rice Premium (5kg)", price: 680, img: "https://images.unsplash.com/photo-1586201375761-83865001e31c?w=200" },
      { id: 4, name: "Tata Tea Gold (1kg)", price: 520, img: "https://images.unsplash.com/photo-1597481499750-3e6b22637e12?w=200" },
      { id: 5, name: "Amul Pure Ghee (1L)", price: 630, img: "https://images.unsplash.com/photo-1628088062854-d1870b4553da?w=200" },
      { id: 6, name: "Dry Fruits Celebration Pack", price: 990, img: "https://images.unsplash.com/photo-1596040033229-a9821ebd058d?w=200" }
    ];

    const rewards = [
      { id: 1, name: "Free 1kg Sugar & Tea Pack", cost: 10, icon: "fa-mug-hot" },
      { id: 2, name: "Non-Stick Fry Pan", cost: 25, icon: "fa-kitchen-set" },
      { id: 3, name: "Electric Kettle (1.5L)", cost: 50, icon: "fa-bolt" },
      { id: 4, name: "Smart Air Fryer / Blender", cost: 100, icon: "fa-dumpster-fire" }
    ];

    let cart = {};
    let userCoupons = parseInt(localStorage.getItem('rai_coupons') || '0');

    function init() {
      renderProducts();
      updateCartUI();
      document.getElementById('userCouponsDisplay').innerText = userCoupons;
    }

    function renderProducts() {
      const grid = document.getElementById('productGrid');
      grid.innerHTML = products.map(p => `
        <div class="bg-white rounded-xl shadow-sm border p-3 flex flex-col justify-between">
          <img src="${p.img}" class="h-28 object-cover rounded-lg mb-2 w-full">
          <div>
            <h4 class="font-bold text-sm text-gray-800 leading-snug">${p.name}</h4>
            <p class="text-green-700 font-extrabold text-base mt-1">₹${p.price}</p>
          </div>
          <button onclick="addToCart(${p.id})" class="mt-3 w-full bg-green-50 text-green-700 hover:bg-green-600 hover:text-white font-bold py-1.5 rounded-lg text-xs transition border border-green-200">
            + Add to Cart
          </button>
        </div>
      `).join('');
    }

    function addToCart(id) {
      cart[id] = (cart[id] || 0) + 1;
      updateCartUI();
    }

    function updateCartUI() {
      const cartItems = document.getElementById('cartItems');
      let total = 0;
      let count = 0;
      let html = '';

      Object.keys(cart).forEach(id => {
        const p = products.find(prod => prod.id == id);
        const qty = cart[id];
        total += p.price * qty;
        count += qty;
        html += `
          <div class="flex items-center justify-between border-b pb-2">
            <div>
              <p class="font-bold text-xs">${p.name}</p>
              <p class="text-xs text-gray-500">₹${p.price} x ${qty}</p>
            </div>
            <div class="flex items-center space-x-2">
              <button onclick="changeQty(${id}, -1)" class="w-6 h-6 border rounded font-bold text-xs">-</button>
              <span class="text-xs font-bold">${qty}</span>
              <button onclick="changeQty(${id}, 1)" class="w-6 h-6 border rounded font-bold text-xs">+</button>
            </div>
          </div>
        `;
      });

      cartItems.innerHTML = html || '<p class="text-center text-gray-400 text-sm py-8">Your cart is empty!</p>';
      document.getElementById('cartTotal').innerText = `₹${total}`;
      document.getElementById('cartBadge').innerText = count;
      document.getElementById('cartBadge').classList.toggle('hidden', count === 0);

      // Coupons Progress
      const couponsEarned = Math.floor(total / 1000);
      const remainder = total % 1000;
      document.getElementById('couponEarnNotice').innerText = `${couponsEarned} Coupon(s)`;
      document.getElementById('couponProgress').style.width = `${(remainder / 1000) * 100}%`;
      document.getElementById('couponSubtext').innerText = total === 0 ? "Add ₹1,000 to earn 1 Coupon!" : `Add ₹${1000 - remainder} more to get another Coupon!`;
    }

    function changeQty(id, delta) {
      cart[id] += delta;
      if (cart[id] <= 0) delete cart[id];
      updateCartUI();
    }

    function toggleCart() { document.getElementById('cartDrawer').classList.toggle('hidden'); }
    function openCheckoutModal() { 
      if (Object.keys(cart).length === 0) return alert("Cart empty hai!");
      document.getElementById('checkoutModal').classList.remove('hidden'); 
    }
    function closeCheckoutModal() { document.getElementById('checkoutModal').classList.add('hidden'); }

    function placeOrder(e) {
      e.preventDefault();
      let total = 0;
      Object.keys(cart).forEach(id => {
        const p = products.find(prod => prod.id == id);
        total += p.price * cart[id];
      });
      const earned = Math.floor(total / 1000);
      userCoupons += earned;
      localStorage.setItem('rai_coupons', userCoupons);
      document.getElementById('userCouponsDisplay').innerText = userCoupons;
      
      alert(`Order Successful (Cash on Delivery)!\nAapne ₹${total} ke order par ${earned} Coupon(s) jeete hain!`);
      cart = {};
      updateCartUI();
      closeCheckoutModal();
      toggleCart();
    }

    function openRewardsModal() {
      document.getElementById('modalCouponBalance').innerText = userCoupons;
      const list = document.getElementById('rewardsList');
      list.innerHTML = rewards.map(r => `
        <div class="border rounded-xl p-3 flex justify-between items-center ${userCoupons >= r.cost ? 'bg-green-50 border-green-300' : 'bg-gray-50'}">
          <div class="flex items-center space-x-3">
            <i class="fa-solid ${r.icon} text-amber-500 text-xl"></i>
            <div>
              <p class="font-bold text-xs">${r.name}</p>
              <p class="text-[10px] text-gray-500">Requires ${r.cost} Coupons</p>
            </div>
          </div>
          <button onclick="redeemReward(${r.cost})" class="px-3 py-1.5 rounded-lg text-xs font-bold ${userCoupons >= r.cost ? 'bg-amber-500 text-white shadow' : 'bg-gray-200 text-gray-400 cursor-not-allowed'}">
            Redeem
          </button>
        </div>
      `).join('');
      document.getElementById('rewardsModal').classList.remove('hidden');
    }

    function closeRewardsModal() { document.getElementById('rewardsModal').classList.add('hidden'); }

    function redeemReward(cost) {
      if (userCoupons < cost) return alert("Aapke paas paryapt coupons nahi hain!");
      userCoupons -= cost;
      localStorage.setItem('rai_coupons', userCoupons);
      document.getElementById('userCouponsDisplay').innerText = userCoupons;
      openRewardsModal();
      alert("Gift Redeem request send ho gayi hai! Dukan se aapko gift mil jayega.");
    }

    init();
  </script>
